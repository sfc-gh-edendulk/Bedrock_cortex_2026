Choose Your Interface

Option A: Snowflake Notebook

Snowflake Notebooks provide an interactive environment for data exploration and AI chat.

Step 1: Create a New Notebook

  1. In Snowsight, click Projects → Notebooks
  2. Click + Notebook
  3. Name it Workshop_AI_Assistant
  4. Select database WORKSHOP_DB, schema AGENTS
  5. Select warehouse WORKSHOP_WH

Step 2: Add Setup Cell

Add a Python cell with the setup code:

# Cell 1: Setup
import streamlit as st
from snowflake.snowpark.context import get_active_session

# Get Snowflake session
session = get_active_session()

# Configuration
AGENT_NAME = "WORKSHOP_DB.AGENTS.WORKSHOP_AGENT"

def ask_agent(question: str) -> str:
    """Send a question to the Cortex Agent"""
    result = session.sql(f"""
        SELECT SNOWFLAKE.CORTEX.AGENT(
            '{AGENT_NAME}',
            '{question.replace("'", "''")}'
        ) AS RESPONSE
    """).collect()
    return result[0]['RESPONSE']

st.title("🤖 Workshop AI Assistant")
st.markdown("Ask questions about FOMC documents or revenue data")

Step 3: Add Chat Interface Cell

# Cell 2: Chat Interface

# Initialize chat history
if "messages" not in st.session_state:
    st.session_state.messages = []

# Display chat history
for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.write(msg["content"])

# Chat input
if prompt := st.chat_input("Ask a question..."):
    # Display user message
    with st.chat_message("user"):
        st.write(prompt)
    st.session_state.messages.append({"role": "user", "content": prompt})
    
    # Get agent response
    with st.chat_message("assistant"):
        with st.spinner("Thinking..."):
            response = ask_agent(prompt)
        st.write(response)
    st.session_state.messages.append({"role": "assistant", "content": response})

Step 5: Run the Notebook

  1. Click Run All to execute all cells
  2. The Streamlit interface will appear in the output
  3. Type questions in the chat input to interact with your agent!

Option B: Streamlit in Snowflake

Deploy a standalone Streamlit app in Snowflake with a polished UI.

Step 1: Create Streamlit App

  1. In Snowsight, click Projects → Streamlit
  2. Click + Streamlit App
  3. Name it Workshop_Chat_App
  4. Select database WORKSHOP_DB, schema AGENTS
  5. Select warehouse WORKSHOP_WH

Step 2: Paste the App Code

Replace the default code with:

import streamlit as st
from snowflake.snowpark.context import get_active_session

# Page config
st.set_page_config(
    page_title="AI Workshop Assistant",
    page_icon="🤖",
    layout="wide"
)

# Get session
session = get_active_session()

# Sidebar
with st.sidebar:
    st.image("https://www.snowflake.com/wp-content/themes/snowflake/assets/img/logo-blue.svg", width=200)
    st.title("Workshop Settings")
    
    agent_type = st.radio(
        "Select Agent:",
        ["Cortex Agent", "Bedrock Agent"],
        index=0
    )
    
    st.markdown("---")
    st.markdown("""
    ### About
    This app demonstrates AI-powered 
    data exploration using:
    - **Cortex Search** for documents
    - **Cortex Analyst** for data
    - **AI Agents** for orchestration
    """)

# Main content
st.title("🤖 Workshop AI Assistant")
st.caption("Ask questions about FOMC documents or revenue data")

# Agent function
def query_agent(question: str, use_cortex: bool = True) -> str:
    if use_cortex:
        result = session.sql(f"""
            SELECT SNOWFLAKE.CORTEX.AGENT(
                'WORKSHOP_DB.AGENTS.WORKSHOP_AGENT',
                $${question}$$
            ) AS RESPONSE
        """).collect()
        return result[0]['RESPONSE']
    else:
        # Bedrock agent call
        result = session.sql(f"""
            SELECT INVOKE_BEDROCK_AGENT($${question}$$) AS RESPONSE
        """).collect()
        return result[0]['RESPONSE']['response']

# Chat history
if "messages" not in st.session_state:
    st.session_state.messages = []

# Display messages
for message in st.session_state.messages:
    with st.chat_message(message["role"]):
        st.markdown(message["content"])

# Chat input
if prompt := st.chat_input("What would you like to know?"):
    st.session_state.messages.append({"role": "user", "content": prompt})
    
    with st.chat_message("user"):
        st.markdown(prompt)
    
    with st.chat_message("assistant"):
        with st.spinner("Analyzing..."):
            use_cortex = agent_type == "Cortex Agent"
            response = query_agent(prompt, use_cortex)
        st.markdown(response)
    
    st.session_state.messages.append({"role": "assistant", "content": response})

# Quick action buttons
st.markdown("---")
st.subheader("Quick Questions")

col1, col2, col3 = st.columns(3)

with col1:
    if st.button("📈 Revenue Summary"):
        st.session_state.messages.append({
            "role": "user",
            "content": "Give me a summary of revenue trends"
        })
        st.rerun()

with col2:
    if st.button("🏛️ Fed Policy"):
        st.session_state.messages.append({
            "role": "user",
            "content": "What is the Fed's current stance on interest rates?"
        })
        st.rerun()

with col3:
    if st.button("🌍 Regional Sales"):
        st.session_state.messages.append({
            "role": "user",
            "content": "Compare revenue across regions"
        })
        st.rerun()

Step 3: Run the App

  1. Click Run in the top right
  2. The app will load in the preview pane
  3. Try toggling between Cortex and Bedrock agents in the sidebar
  4. Ask questions using the chat input or quick buttons!
✅ Streamlit App Complete! You now have a polished chat application.
